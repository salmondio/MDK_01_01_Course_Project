CREATE TABLE Role
(
	Id INT PRIMARY KEY IDENTITY(1, 1),
	Name VARCHAR(100) NOT NULL,
	Description VARCHAR(1000)
);

CREATE TABLE User
(
	Id INT PRIMARY KEY IDENTITY(1, 1),
	Id_role INT NOT NULL,
	Name NVARCHAR(100) NOT NULL,
	Lastname NVARCHAR(100) NOT NULL,
	Surname NVARCHAR(100),
	Email VARCHAR(100) NOT NULL,
	Phone_number DECIMAL(11, 0),
	Password VARCHAR(256) NOT NULL,
	Token VARCHAR(256),
	Is_active BOOLEAN NOT NULL DEFAULT TRUE,

	FOREIGN KEY(Id_role) REFERENCES Role(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION
);

CREATE TABLE Discipline
(
	Id INT PRIMARY KEY IDENTITY(1, 1),
	Name NVARCHAR(100) NOT NULL,
	Description NVARCHAR(1000)
);

CREATE TABLE Teacher_discipline
(
	Id INT PRIMARY KEY IDENTITY(1, 1),
	Id_teacher INT NOT NULL,
	Id_discipline INT NOT NULL,
	
	FOREIGN KEY(Id_teacher) REFERENCES User(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION,
	
	FOREIGN KEY(Id_discipline) REFERENCES Disciplines(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION
);

CREATE TABLE Message_status
(
	Id INT PRIMARY KEY IDENTITY(1, 1),
	Name NVARCHAR(100),
	Description NVARCHAR(1000)
);

CREATE TABLE Review
(
	Id INT PRIMARY KEY IDENTITY(1, 1),
	Id_student INT NOT NULL,
	Id_teacher INT NOT NULL,
	Id_status INT NOT NULL,
	Id_inspector INT,
	Text NVARCHAR(MAX),
	
	FOREIGN KEY(Id_student) REFERENCES User(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION,
	
	FOREIGN KEY(Id_teacher) REFERENCES User(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION
	
	FOREIGN KEY(Id_status) REFERENCES Status(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION,
	
	FOREIGN KEY(Id_inspector) REFERENCES User(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION
);

CREATE TABLE Report
(
	Id INT PRIMARY KEY IDENTITY(1, 1),
	Id_student INT NOT NULL,
	Id_teacher INT NOT NULL,
	Id_status INT NOT NULL,
	Id_inspector INT,
	Text NVARCHAR(MAX),
	
	FOREIGN KEY(Id_student) REFERENCES User(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION,
	
	FOREIGN KEY(Id_teacher) REFERENCES User(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION
	
	FOREIGN KEY(Id_status) REFERENCES Status(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION,
	
	FOREIGN KEY(Id_inspector) REFERENCES User(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION
);

CREATE TABLE Evaluation
(
	Id INT PRIMARY KEY IDENTITY(1, 1),
	Id_student INT NOT NULL,
	Id_teacher INT NOT NULL,
	Presentation TINYINT NOT NULL CHECK (Presentation BETWEEN 1 AND 9),
	Attitude TINYINT NOT NULL CHECK (Presentation BETWEEN 1 AND 9),
	Responsiveness TINYINT NOT NULL CHECK (Presentation BETWEEN 1 AND 9),
	
	CONSTRAINT FK_Evaluation_Student FOREIGN KEY(Id_student) REFERENCES User(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION,
	
	CONSTRAINT FK_Evaluation_Teacher FOREIGN KEY(Id_teacher) REFERENCES User(Id)
	ON UPDATE CASCADE
	ON DELETE NO ACTION
);

CREATE INDEX IX_Evaluation_StudentId ON Evaluation(Id_student);
CREATE INDEX IX_Evaluation_TeacherId ON Evaluation(Id_teacher);
